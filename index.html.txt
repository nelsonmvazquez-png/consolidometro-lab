        <label for="presion">Presión deseada (kg/cm²):</label>
        <input type="number" id="presion" value="0.05" step="0.001">

        <button onclick="calcular()" style="margin-top:15px; width:100%; padding:10px; background:#007bff; color:white; border:none; border-radius:4px; font-size:16px;">Calcular Peso</button>

            <li>Verificar que el brazo de palanca esté nivelado y equilibrado antes de colocar la muestra.</li>
            <li>Colocar la celda de consolidación centrada bajo el punto de compresión.</li>
            <li>Ajustar el deformatímetro (dial) a cero.</li>
            <li>Aplicar la carga calculada en el plato suspendido.</li>
        </ol>

    <script>
        function calcular() {
            let A = parseFloat(document.getElementById('area').value);
            let P = parseFloat(document.getElementById('presion').value);
            let R = 76.0 / 6.0; // Relación de palanca exacta (12.6667)
            
            if (A > 0 && P > 0) {
                let fuerza = P * A;
                let cargaKg = fuerza / R;
                let cargaGramo = cargaKg * 1000;
                
                document.getElementById('resultado').innerHTML = 
                    `Fuerza sobre muestra: ${fuerza.toFixed(2)} kg<br>` +
                    `Carga en plato: <b>${cargaKg.toFixed(3)} kg (${cargaGramo.toFixed(1)} g)</b>`;
            } else {
                document.getElementById('resultado').innerHTML = "Por favor ingrese valores válidos.";
            }
        }
    </script>
